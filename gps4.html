function generateCode() {
const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
let code = "";
for (let i = 0; i < 6; i++) {
code += chars.charAt(Math.floor(Math.random() * chars.length));
}
return code;
}

let myCode = localStorage.getItem("myTrackCode");
if (!myCode) {
myCode = generateCode();
localStorage.setItem("myTrackCode", myCode);
}

let pairedCode = (localStorage.getItem("pairedTrackCode") || "").toUpperCase();

myCodeEl.textContent = myCode;
friendCodeInput.value = pairedCode;

function setStatus(msg) {
statusMsgEl.textContent = msg;
}

function refreshStatusFromState(data) {
if (pairedCode && data && data[pairedCode]) {
setStatus(t("liveTracking", pairedCode));
} else if (pairedCode) {
setStatus(t("waitingFor", pairedCode));
} else {
setStatus(t("myCodeReady", myCode));
}
}

const markers = {};

function renderUsers(usersObj) {
userListEl.innerHTML = "";
const activeIds = [];

if (myCode && usersObj && usersObj[myCode]) {
activeIds.push(myCode);
}
if (pairedCode && pairedCode !== myCode && usersObj && usersObj[pairedCode]) {
activeIds.push(pairedCode);
}

if (activeIds.length === 0) {
userListEl.innerHTML = `<div class="user-card"><div>${t("noUsers")}</div></div>`;
} else {
activeIds.forEach((id) => {
const u = usersObj[id];
if (!u || typeof u.lat !== "number" || typeof u.lng !== "number") return;

const speedKmh = typeof u.speed === "number" && u.speed >= 0
? (u.speed * 3.6).toFixed(1)
: "0.0";

const label = id === myCode ? `${t("me")} (${id})` : `${t("friend")} (${id})`;

const card = document.createElement("div");
card.className = "user-card";
card.innerHTML = `
<div class="user-meta">
<strong>${label}</strong>
<small>${u.lat.toFixed(5)}, ${u.lng.toFixed(5)}</small>
</div>
<div class="speed-badge">${speedKmh} km/h</div>
`;
card.addEventListener("click", () => {
map.setView([u.lat, u.lng], 15);
});
userListEl.appendChild(card);

if (!markers[id]) {
markers[id] = L.marker([u.lat, u.lng]).addTo(map).bindPopup(label);
} else {
markers[id].setLatLng([u.lat, u.lng]).bindPopup(label);
}
});
}

Object.keys(markers).forEach((id) => {
if (!activeIds.includes(id)) {
map.removeLayer(markers[id]);
delete markers[id];
}
});
}

connectBtn.addEventListener("click", () => {
const code = friendCodeInput.value.replace(/[^A-Za-z0-9]/g, "").toUpperCase();
friendCodeInput.value = code;
if (!code) {
setStatus(t("enterFriendCode"));
return;
}
if (code === myCode) {
setStatus(t("ownCodeError"));
return;
}
pairedCode = code;
localStorage.setItem("pairedTrackCode", pairedCode);
update(ref(db, `users/${myCode}`), { pairedWith: pairedCode }).catch(() => {});
setStatus(t("connectingTo", pairedCode));
renderUsers(latestUsersData);
});

logoutBtn.addEventListener("click", async () => {
if (!confirm(t("logoutConfirm"))) return;

const oldCode = myCode;
if (oldCode) {
try {
await remove(ref(db, `users/${oldCode}`));
} catch (_) {}
}

const newCode = generateCode();
myCode = newCode;
pairedCode = "";

localStorage.setItem("myTrackCode", myCode);
localStorage.removeItem("pairedTrackCode");

myCodeEl.textContent = myCode;
friendCodeInput.value = "";

if (lastPos) {
const { latitude, longitude, speed, heading } = lastPos.coords;
set(ref(db, `users/${myCode}`), {
lat: latitude,
lng: longitude,
speed: speed ?? 0,
heading: heading ?? 0,
pairedWith: "",
updatedAt: Date.now()
}).catch(() => {});
}

renderUsers(latestUsersData);
setStatus(t("loggedOutMsg", myCode));
});

onValue(
ref(db, "users"),
(snapshot) => {
const data = snapshot.val() || {};
latestUsersData = data;
renderUsers(data);
refreshStatusFromState(data);
},
(err) => {
setStatus(t("fbReadError", err.message));
}
);

if ("geolocation" in navigator) {
navigator.geolocation.watchPosition(
(pos) => {
lastPos = pos;
const { latitude, longitude, speed, heading } = pos.coords;
set(ref(db, `users/${myCode}`), {
lat: latitude,
lng: longitude,
speed: speed ?? 0,
heading: heading ?? 0,
pairedWith: pairedCode || "",
updatedAt: Date.now()
}).catch((err) => {
setStatus(t("fbSaveError", err.message));
});
},
(err) => {
setStatus(t("gpsError", err.message));
},
{
enableHighAccuracy: true,
maximumAge: 2000,
timeout: 10000
}
);
} else {
setStatus(t("noGeo"));
}

applyLanguage();
</script>
</body>
